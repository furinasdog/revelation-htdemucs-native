cmake_minimum_required(VERSION 3.21)
project(revelation_htdemucs VERSION 0.1.0 LANGUAGES CXX)

if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This executable currently targets 64-bit Windows.")
endif()
set(ONNXRUNTIME_DIR "" CACHE PATH "Local ONNX Runtime C/C++ SDK with include/ and lib/")
if(ONNXRUNTIME_DIR)
  set(ort_headers "${ONNXRUNTIME_DIR}/include")
  set(ort_library_dir "${ONNXRUNTIME_DIR}/lib")
else()
  include(FetchContent)
  FetchContent_Declare(ort_sdk
    URL https://api.nuget.org/v3-flatcontainer/microsoft.ml.onnxruntime/1.23.2/microsoft.ml.onnxruntime.1.23.2.nupkg
    URL_HASH SHA256=25fe172f7fcdf34f2b5b02c8b997f6b88abc282956849d4af4335af23d0c4e4a)
  FetchContent_MakeAvailable(ort_sdk)
  set(ort_headers "${ort_sdk_SOURCE_DIR}/build/native/include")
  set(ort_library_dir "${ort_sdk_SOURCE_DIR}/runtimes/win-x64/native")
endif()

add_executable(revelation-stems src/main.cpp src/process.cpp src/separate.cpp)
target_compile_features(revelation-stems PRIVATE cxx_std_17)
target_compile_definitions(revelation-stems PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(revelation-stems PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_include_directories(revelation-stems PRIVATE "${ort_headers}")
target_link_libraries(revelation-stems PRIVATE "${ort_library_dir}/onnxruntime.lib")
add_custom_command(TARGET revelation-stems POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${ort_library_dir}/onnxruntime.dll" "$<TARGET_FILE_DIR:revelation-stems>")
install(TARGETS revelation-stems RUNTIME DESTINATION .)
install(FILES "${ort_library_dir}/onnxruntime.dll" DESTINATION .)
install(FILES LICENSE THIRD_PARTY.md licenses/Demucs-MIT.txt DESTINATION licenses)
if(NOT ONNXRUNTIME_DIR)
  install(FILES "${ort_sdk_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME ONNXRuntime-LICENSE.txt)
  install(FILES "${ort_sdk_SOURCE_DIR}/ThirdPartyNotices.txt" DESTINATION licenses RENAME ONNXRuntime-ThirdPartyNotices.txt)
endif()
